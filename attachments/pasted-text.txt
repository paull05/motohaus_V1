/* =========================================================================
   STORAGE LAYER — this app has no real server. All "backend" state lives in
   the artifact's per-user persistent key/value storage (window.storage),
   which survives reloads for the same user. Swap these functions for real
   fetch() calls to an API and the rest of the app is unaffected.
   ========================================================================= */
async function kvGet(key){
  try{ const r = await window.storage.get(key, false); return r ? JSON.parse(r.value) : null; }
  catch(e){ return null; }
}
async function kvSet(key, value){
  try{ await window.storage.set(key, JSON.stringify(value), false); return true; }
  catch(e){ return false; }
}

function filterOrders(list){
  let out = list;
  if(ui.searchQuery){
    out = out.filter(o => (o.id+o.customer+o.item).toLowerCase().includes(ui.searchQuery));
  }
  if(ui.statusFilter !== 'all'){
    out = out.filter(o => o.status === ui.statusFilter);
  }
  if(ui.sortBy === 'cost'){
    out = [...out].sort((a,b)=>(a.cost-b.cost)*ui.sortDir);
  } else if(ui.sortBy === 'due'){
    out = [...out].sort((a,b)=>(new Date(a.due)-new Date(b.due))*ui.sortDir);
  }
  return out;
}
function filterCustomers(list){
  if(!ui.searchQuery) return list;
  return list.filter(c => (c.name+c.email).toLowerCase().includes(ui.searchQuery));
}

document.getElementById('orderForm').addEventListener('submit', async (e)=>{
  e.preventDefault();
  const editId = document.getElementById('orderEditId').value;
  const cost = parseFloat(document.getElementById('orderCost').value);
  if(isNaN(cost) || cost < 0){
    document.getElementById('orderFormError').textContent = 'Enter a valid cost.'; return;
  }
  const data = {
    customer: document.getElementById('orderCustomer').value.trim(),
    item: document.getElementById('orderItem').value.trim(),
    cost: cost,
    due: document.getElementById('orderDue').value,
    payment: document.getElementById('orderPayment').value,
    status: document.getElementById('orderStatus').value,
  };
  const btn = document.getElementById('orderSubmitBtn');
  btn.disabled = true;
  document.getElementById('orderSubmitLabel').innerHTML = '<span class="spinner"></span>';
  await new Promise(r=>setTimeout(r,350));

  if(editId){
    const idx = orders.findIndex(o=>o.id===editId);
    if(idx>-1){
      const oldCost = orders[idx].cost;
      orders[idx] = {...orders[idx], ...data};
      metrics.baseSale += (data.cost - oldCost);
    }
    toast('Order updated.', 'success');
  } else {
    const newId = 'ORD' + Math.floor(100000+Math.random()*899999);
    orders.push({id:newId, ...data});
    metrics.baseOrders += 1;
    metrics.baseSale += data.cost;
    toast('Order added.', 'success');
  }
  await kvSet('orders', orders);
  await kvSet('metrics', metrics);
  btn.disabled = false;
  closeModal('orderModalOverlay');
  renderAll();
});

/* ---- SVG chart drawing (kept visually identical to the original design) ---- */
function pathFor(arr, max, w, h){
  const stepX = w/(arr.length-1);
  const pts = arr.map((v,i)=>[i*stepX, h - (v/max)*h*0.92 - 6]);
  let d = `M ${pts[0][0]},${pts[0][1]}`;
  for(let i=1;i<pts.length;i++){
    const [x0,y0]=pts[i-1], [x1,y1]=pts[i];
    const mx=(x0+x1)/2;
    d += ` C ${mx},${y0} ${mx},${y1} ${x1},${y1}`;
  }
  return d;
}
function drawAreaChart(elId, revenue, orders, w, h, hlIndex){
  const max = Math.max(...revenue, ...orders, 1) * 1.15;
  const stepX = w/(revenue.length-1);
  const revPath = pathFor(revenue, max, w, h);
  const ordPath = pathFor(orders, max, w, h);
  const barW = stepX*0.42;
  let bars = '';
  orders.forEach((v,i)=>{
    const x = i*stepX - barW/2;
    const bh = (v/max)*h*0.85;
    const y = h - bh - 6;
    bars += `<rect x="${x}" y="${y}" width="${barW}" height="${bh}" rx="3" fill="${i===hlIndex ? 'url(#gradOrangeBar)' : 'rgba(139,107,255,0.18)'}"></rect>`;
  });
  document.getElementById(elId).innerHTML = `
  <svg viewBox="0 0 ${w} ${h}" preserveAspectRatio="none">
    <defs>
      <linearGradient id="gradOrangeBar" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stop-color="#ff6a3d"/><stop offset="100%" stop-color="#ff2d55"/></linearGradient>
      <linearGradient id="gradRevFill" x1="0" y1="0" x2="0" y2="1"><stop offset="0%" stop-color="#ff6a3d" stop-opacity="0.35"/><stop offset="100%" stop-color="#ff6a3d" stop-opacity="0"/></linearGradient>
    </defs>
    ${bars}
    <path d="${revPath} L ${w},${h} L 0,${h} Z" fill="url(#gradRevFill)"></path>
    <path d="${revPath}" fill="none" stroke="#ff6a3d" stroke-width="2.2"></path>
    <path d="${ordPath}" fill="none" stroke="#8b6bff" stroke-width="2" opacity="0.85"></path>
  </svg>`;
}
function drawSparkline(elId, a, b, w, h){
  const max = Math.max(...a, ...b) * 1.15;
  const stepX = w/(a.length-1);
  function pf(arr){
    const pts = arr.map((v,i)=>[i*stepX, h-(v/max)*h*0.85-4]);
    let d = `M ${pts[0][0]},${pts[0][1]}`;
    for(let i=1;i<pts.length;i++){ const [x0,y0]=pts[i-1],[x1,y1]=pts[i]; const mx=(x0+x1)/2; d += ` C ${mx},${y0} ${mx},${y1} ${x1},${y1}`; }
    return d;
  }
  document.getElementById(elId).innerHTML = `
  <svg viewBox="0 0 ${w} ${h}" preserveAspectRatio="none">
    <path d="${pf(a)}" fill="none" stroke="#ff6a3d" stroke-width="2"></path>
    <path d="${pf(b)}" fill="none" stroke="#8b6bff" stroke-width="2"></path>
    <line x1="${4*stepX}" y1="0" x2="${4*stepX}" y2="${h}" stroke="#ff6a3d" stroke-width="1" stroke-dasharray="2 3" opacity="0.5"/>
    <circle cx="${4*stepX}" cy="${h-(a[4]/max)*h*0.85-4}" r="3.5" fill="#ff6a3d"/>
  </svg>`;
}

  document.querySelectorAll('[data-cart-inc]').forEach(el=>el.addEventListener('click',()=>{const item=shopCart.find(i=>i.id===el.dataset.cartInc),p=shopProduct(item.id);if(item.qty<p.stock)item.qty++;shopRenderPage();}));
  document.querySelectorAll('[data-cart-dec]').forEach(el=>el.addEventListener('click',()=>{const item=shopCart.find(i=>i.id===el.dataset.cartDec);item.qty--;if(item.qty<1)shopCart=shopCart.filter(i=>i.id!==item.id);shopRenderPage();}));
  document.querySelectorAll('[data-product-view]').forEach(el=>el.addEventListener('click',()=>shopOpenModal('productView',el.dataset.productView)));
  document.querySelectorAll('[data-product-edit]').forEach(el=>el.addEventListener('click',()=>shopOpenModal('product',el.dataset.productEdit)));
  document.querySelectorAll('[data-customer-view]').forEach(el=>el.addEventListener('click',()=>shopOpenModal('customerView',el.dataset.customerView)));
  document.querySelectorAll('[data-supplier-view]').forEach(el=>el.addEventListener('click',()=>shopOpenModal('supplierView',el.dataset.supplierView)));
  document.querySelectorAll('[data-receive]').forEach(el=>el.addEventListener('click',()=>{const p=shopState.purchases.find(x=>x.id===el.dataset.receive);p.status='Received';shopState.audit.unshift({action:'Purchase received',detail:p.id,user:shopUser.name,date:'2026-09-05 09:30'});shopSave();shopRenderPage();toast('Purchase received and inventory is ready for update.','success');}));
  const checkout=document.querySelector('[data-checkout]');if(checkout)checkout.addEventListener('click',()=>shopOpenModal('checkout'));
  const clear=document.querySelector('[data-cart-clear]');if(clear)clear.addEventListener('click',()=>{shopCart=[];shopRenderPage();});
}

function shopModalSubmit(type,id){const form=document.getElementById('shopModalForm');if(!form||!form.reportValidity())return;const data=Object.fromEntries(new FormData(form));if(type==='product'){const p={id:id||'P-'+Date.now().toString().slice(-5),sku:data.sku,name:data.name,category:data.category,brand:data.brand,compatibility:data.compatibility,supplier:'RiderPro Supply',price:Number(data.price),cost:Number(data.price)*.62,stock:Number(data.stock),min:Number(data.min),location:data.location,description:'Demo catalog part.'};if(id)shopState.products=shopState.products.map(x=>x.id===id?Object.assign(x,p):x);else shopState.products.unshift(p);shopState.audit.unshift({action:id?'Product updated':'Product added',detail:p.name,user:shopUser.name,date:'2026-09-05 09:30'});}
  if(type==='customer'){const c={id:id||'C-'+String(shopState.customers.length+1).padStart(3,'0'),name:data.name,phone:data.phone,email:data.email,motorcycle:data.motorcycle,purchases:0,last:'—'};if(id)shopState.customers=shopState.customers.map(x=>x.id===id?Object.assign(x,c):x);else shopState.customers.push(c);}
  if(type==='supplier'){const s={id:id||'S-'+String(shopState.suppliers.length+1).padStart(3,'0'),company:data.company,contact:data.contact,phone:data.phone,email:data.email,status:'Active',products:0,purchases:0};if(id)shopState.suppliers=shopState.suppliers.map(x=>x.id===id?Object.assign(x,s):x);else shopState.suppliers.push(s);}
  if(type==='stock'){const p=shopProduct(data.productId);p.stock+=Number(data.qty);shopState.audit.unshift({action:'Stock received',detail:p.name,user:shopUser.name,date:'2026-09-05 09:30'});}
  if(type==='checkout'){const total=shopCart.reduce((a,i)=>a+i.qty*i.price,0);shopCart.forEach(i=>{const p=shopProduct(i.id);p.stock-=i.qty;});const txn={id:'TXN-'+Date.now().toString().slice(-8),customer:data.customer,items:shopCart.map(i=>({productId:i.id,qty:i.qty})),total,payment:data.payment,staff:shopUser.name,date:'2026-09-05'};shopState.sales.unshift(txn);shopState.audit.unshift({action:'Sale completed',detail:txn.id,user:shopUser.name,date:'2026-09-05 09:30'});shopCart=[];}
  shopSave();document.getElementById('shopModalRoot').innerHTML='';shopRenderPage();toast(type==='checkout'?'Sale completed and inventory updated.':'Saved successfully.','success');
}
